CREATE TABLE & INSERT Data

Teori

SQL (Structured Query Language) adalah bahasa untuk mengelola database. CREATE TABLE membuat tabel baru dengan mendefinisikan kolom dan tipe datanya. INSERT INTO memasukkan baris data ke tabel. Tipe data umum: VARCHAR (teks), INT (bilangan bulat), DECIMAL (desimal), DATE (tanggal), BOOLEAN (benar/salah).
